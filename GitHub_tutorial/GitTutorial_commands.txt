Commands

git status
git add
git commit

git log
git reset
  git reset --hard <hash>
  git reset --soft HEAD~1
  git stash
  git stash pop
  git revert

git branch
  git branch [-v] <<to list branches>>
  git branch <branch>
  git checkout <branch>
  git checkout -b <new_branch>
  git branch -D <branch>

git checkout
  git checkout -- <file>

git merge

git push
  git push --set-upstream origin <branch>  <<same as -u>>
  git push --force-with-lease
git pull
git remote add origin <url>

git init
git clone
  ALT: via Github Desktop

<<Creating a Repo>>
mkdir ghtutorial-2026
cd ghtutorial-2026
echo "# README: ghtutorial-2026" >> README.md
git init
git add README.md
git commit -m "first commit"

<<Syncing to GitHub remote>>
git remote add origin https://github.com/<username>/ghtutorial-2026.git
git push --set-upstream origin main
